//pass function as a parameter
(func add3 (a b c)
  (plus (plus a b ) c))
(func doAdd3 (lst b c)
  (lst b c -2))
(doAdd3 add3 2 5) //8

(func doAdd3v2 (fn arg1 arg2)
  (fn arg1 arg2 -2))
(doAdd3v2 (lambda (a1 a2 a3) (plus (plus a1 a2) a3)) 2 5) //8


//evaluating a func then calls it
(func negative (a) (
  (lambda (a) (minus 0 a))
  a
))
(negative -5) // 5
